## Iterative code
```cpp 
#include <bits/stdc++.h>

void solve() {
	int n , totW;
	cin >> n >> totW ;
	vector<pair<int,int>>p(n);
	for( int i = 0 ; i < n ; i++ ){
		// w val
		cin >> p[i].first >> p[i].second;
	}
	
	// i = number of items considered
    // w = current capacity
	
	vector<vector<int>> dp(n+1, vector<int>(totW+1, 0)) ;
	for( int i = 1 ; i <= n ; i++ ){ 
		for ( int w = 1 ; w <= totW ; w++){
			if ( p[i-1].first <= w ){
				int take = p[i-1].second + dp[i-1][w - p[i-1].first];
				int notTake = dp[i-1][w];
				dp[i][w] = max ( take , notTake);
			}
			else{
				dp[i][w] = dp[i-1][w];
			}
		}
	}
	cout << dp[n][totW] << '\n';
}

int32_t main() {
	ios_base::sync_with_stdio(false);
	cin.tie(NULL);
	
	int tests = 1;
	//cin >> tests;
	for ( int tc = 1; tc <= tests; tc++ ){
		solve();
	}
	return 0;
}
```

# Recursive Code 
```cpp
recursive approach
const int mx = 1e5+12;
int n , totW;
int dp[100+12][mx];

int yo ( int idx , vector<pair<int,int>> &p , int curW){
	if ( idx >= n ) return 0 ;
	if (dp[idx][curW] != -1) return dp[idx][curW];
	
	int notTake = 0 ;
	notTake += yo(idx+1, p , curW);
	int take = 0;
	if ( curW + p[idx].first <= totW){
		take += yo( idx+1, p , curW+p[idx].first) + p[idx].second;
	}
	return dp[idx][curW] = max( take , notTake);
}
```
